---------------------------------------------------hashlib------------------------------------------------------

import hashlib
texte = "Bonjour Patrice !"
empreinte = hashlib.sha256(texte.encode()).hexdigest()
print(empreinte)

1 bit       = 0 ou 1
4 bits      = 1 chiffre hexadécimal
8 bits      = 2 chiffres hexadécimaux = 1 octet
256 bits    = 64 chiffres hexadécimaux = 32 octets
